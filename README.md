/* ================= WORMHOLE INTRO ================= */
(function(){
  const ov=document.getElementById("warp"),big=document.getElementById("bigName");
  const reduce=matchMedia("(prefers-reduced-motion: reduce)").matches;
  if(!ov||reduce){ov&&ov.remove();big&&big.remove();document.body.classList.add("arrived");return;}
  document.body.classList.add("warping");
  const cv=document.getElementById("warpCanvas"),ctx=cv.getContext("2d"),flash=document.getElementById("warpFlash");
  let W,H,dpr,S,mx=0,my=0,tx=0,ty=0,done=false,raf,popped=false;
  function size(){dpr=Math.min(devicePixelRatio||1,2);W=innerWidth;H=innerHeight;cv.width=W*dpr;cv.height=H*dpr;ctx.setTransform(dpr,0,0,dpr,0,0);S=Math.min(W,H);}
  size();addEventListener("resize",size);
  addEventListener("pointermove",e=>{tx=(e.clientX/W-.5)*2;ty=(e.clientY/H-.5)*2;},{passive:true});
  const N=W<700?560:1100;
  const mk=init=>({a:Math.random()*6.2832,rho:.1+Math.pow(Math.random(),.65)*1.15,z:init?Math.random():1,w:Math.random(),t:Math.random()});
  const P=Array.from({length:N},()=>mk(true));
  const RINGS=16;
  let ringPhase=0,t0=performance.now(),last=t0,spin=0;
  /* timeline (ms): fly -> white spot grows -> white light fills screen -> big name pops -> site arrives */
  const SPOT=3100,WHITE=4500,ARRIVE=5700,TOTAL=7600;
  const ease=x=>x*x*(3-2*x),clamp=x=>Math.max(0,Math.min(1,x));
  function proj(a,rho,z,cx,cy,sp){const r=rho/z*S*.17,aa=a+sp+(1-z)*.9;return[cx+Math.cos(aa)*r,cy+Math.sin(aa)*r*.94,r];}

  function frame(now){
    if(done)return;
    raf=requestAnimationFrame(frame);
    const t=now-t0,dt=Math.min(.05,(now-last)/1000);last=now;
    mx+=(tx-mx)*.04;my+=(ty-my)*.04;
    const k=Math.min(1,t/3200),speed=.16+2.5*Math.pow(ease(k),1.5);
    spin+=dt*.12*(1+speed);
    const cx=W/2+mx*S*.04,cy=H/2+my*S*.03;
    const e=clamp((t-SPOT)/(WHITE-SPOT));

    ctx.globalCompositeOperation="source-over";
    ctx.fillStyle="rgba(0,0,0,.30)";ctx.fillRect(0,0,W,H);
    ctx.globalCompositeOperation="lighter";
    ringPhase=(ringPhase+dt*speed*.18)%1;
    for(let i=0;i<RINGS;i++){
      const z=1-((i/RINGS+ringPhase)%1)*.985;
      const r=S*.17*.62/z,al=Math.pow(1-z,1.4)*.22*(1-e*.6);
      if(r>Math.hypot(W,H))continue;
      ctx.save();ctx.translate(cx,cy);ctx.rotate(spin*1.4+(1-z)*.9);
      ctx.strokeStyle=`rgba(${130+((1-z)*70|0)},${110+((1-z)*50|0)},255,${al})`;
      ctx.lineWidth=1+(1-z)*5;ctx.setLineDash([r*.55,r*.35,r*.12,r*.4]);
      ctx.beginPath();ctx.ellipse(0,0,r,r*.94,0,0,6.2832);ctx.stroke();ctx.restore();
    }
    ctx.setLineDash([]);
    const dz=dt*speed*.22;
    for(const p of P){
      const pz=p.z+dz*2.2;p.z-=dz;
      if(p.z<=.02){Object.assign(p,mk(false));continue;}
      const[x1,y1,r1]=proj(p.a,p.rho,p.z,cx,cy,spin),[x0,y0]=proj(p.a,p.rho,Math.min(1,pz),cx,cy,spin);
      if(r1>Math.hypot(W,H)){Object.assign(p,mk(false));continue;}
      const near=1-p.z,al=Math.min(1,near*1.8+.05);
      const warm=p.w<.55?`${185+(p.w*60|0)},${160+(p.w*70|0)},255`:p.w<.85?"255,255,255":"150,185,255";
      ctx.strokeStyle=`rgba(${warm},${al})`;ctx.lineWidth=.5+near*near*2.6*(p.t>.9?1.7:1);
      ctx.beginPath();ctx.moveTo(x0,y0);ctx.lineTo(x1,y1);ctx.stroke();
    }
    /* black hole with a purple glow ring */
    const hr=S*.045*(1+e*e*5);
    const glow=ctx.createRadialGradient(cx,cy,hr*.85,cx,cy,hr*4.2);
    glow.addColorStop(0,"rgba(236,230,255,.95)");glow.addColorStop(.18,"rgba(150,130,255,.58)");glow.addColorStop(.5,"rgba(110,80,255,.18)");glow.addColorStop(1,"rgba(90,60,255,0)");
    ctx.fillStyle=glow;ctx.beginPath();ctx.arc(cx,cy,hr*4.2,0,6.2832);ctx.fill();
    ctx.save();ctx.translate(cx,cy);ctx.scale(1,.075);
    const disc=ctx.createRadialGradient(0,0,hr*.9,0,0,hr*7.2);
    disc.addColorStop(0,"rgba(244,240,255,.9)");disc.addColorStop(.25,"rgba(170,145,255,.55)");disc.addColorStop(1,"rgba(110,80,255,0)");
    ctx.fillStyle=disc;ctx.beginPath();ctx.arc(0,0,hr*7.2,0,6.2832);ctx.fill();ctx.restore();
    ctx.globalCompositeOperation="source-over";
    ctx.fillStyle="#000";ctx.beginPath();ctx.arc(cx,cy,hr*.9,0,6.2832);ctx.fill();
    ctx.strokeStyle="rgba(238,232,255,.95)";ctx.lineWidth=Math.max(1.2,hr*.035);ctx.beginPath();ctx.arc(cx,cy,hr*.92,0,6.2832);ctx.stroke();
    /* the white spot: starts at the centre, swells, then white light floods everything */
    if(e>0){
      const R=Math.pow(e,1.7)*Math.hypot(W,H)*.85;
      const g=ctx.createRadialGradient(cx,cy,0,cx,cy,R);
      g.addColorStop(0,"rgba(255,255,255,1)");g.addColorStop(.55,"rgba(240,234,255,.96)");g.addColorStop(1,"rgba(170,150,255,0)");
      ctx.fillStyle=g;ctx.beginPath();ctx.arc(cx,cy,R,0,6.2832);ctx.fill();
    }
    flash.style.opacity=ease(clamp((e-.55)/.45)).toFixed(3);

    if(t>WHITE-150&&!popped){popped=true;big.classList.add("pop");}           // big ARID-X pops out in the white light
    if(t>WHITE&&!ov.classList.contains("out"))ov.classList.add("out");         // white fades away into the site
    if(t>ARRIVE&&!document.body.classList.contains("arrived")){document.body.classList.remove("warping");document.body.classList.add("arrived");}
    if(t>TOTAL)finish();
  }
  function finish(){
    if(done)return;done=true;cancelAnimationFrame(raf);
    document.body.classList.remove("warping");document.body.classList.add("arrived");
    ov.classList.add("out");big.style.display="none";setTimeout(()=>{ov.remove();big.remove();},1200);
  }
  document.getElementById("warpSkip").addEventListener("click",finish);
  addEventListener("keydown",e=>{if(!done&&(e.key==="Escape"||e.key==="Enter"||e.key===" "))finish();});
  setTimeout(finish,10000);                       // failsafe: never trap the visitor
  raf=requestAnimationFrame(f=>{t0=last=f;frame(f);});
})();

/* ================= STARFIELD ================= */
(function(){
  const cv=document.getElementById("stars"),ctx=cv.getContext("2d");
  const reduce=matchMedia("(prefers-reduced-motion: reduce)").matches;
  let W,H,dpr,stars=[],shoot=null,nextShoot=0,mx=0,my=0,sy=0,last=0,raf;
  const LAYERS=[{speed:.012,par:.01},{speed:.025,par:.025},{speed:.045,par:.05}];
  function resize(){
    dpr=Math.min(devicePixelRatio||1,2);W=innerWidth;H=innerHeight;
    cv.width=W*dpr;cv.height=H*dpr;ctx.setTransform(dpr,0,0,dpr,0,0);
    const n=W<768?120:Math.min(320,Math.round(W*H/5500)+150);
    stars=Array.from({length:n},()=>({x:Math.random()*W,y:Math.random()*H,r:.5+Math.random()*1.5,
      o:.2+Math.random()*.8,l:Math.floor(Math.random()*3),ph:Math.random()*6.28,sp:.4+Math.random()*1.2}));
    draw(0);
  }
  function draw(t){
    ctx.clearRect(0,0,W,H);
    // nebula glow behind hero
    const g=ctx.createRadialGradient(W/2,H*.25-sy*.2,0,W/2,H*.25-sy*.2,Math.max(W,H)*.45);
    g.addColorStop(0,"rgba(124,140,255,.10)");g.addColorStop(1,"rgba(124,140,255,0)");
    ctx.fillStyle=g;ctx.fillRect(0,0,W,H);
    const s=t/1000;
    for(const st of stars){
      const L=LAYERS[st.l];
      let x=st.x,y=st.y;
      if(!reduce){
        x=(st.x+s*L.speed*60+mx*L.par*30)%W;if(x<0)x+=W;
        y=((st.y-sy*L.par*.6+s*L.speed*8+my*L.par*30)%H+H)%H;
      }
      const tw=reduce?1:.55+.45*Math.sin(s*st.sp+st.ph);
      ctx.globalAlpha=Math.max(.05,st.o*tw);
      ctx.fillStyle="#fff";ctx.beginPath();ctx.arc(x,y,st.r,0,6.283);ctx.fill();
    }
    ctx.globalAlpha=1;
    if(!reduce){
      if(!shoot&&s>nextShoot){
        const a=Math.PI/5+Math.random()*.3;
        shoot={x:Math.random()*W*.8,y:Math.random()*H*.4,vx:Math.cos(a)*9,vy:Math.sin(a)*9,life:0};
        nextShoot=s+6+Math.random()*6;
      }
      if(shoot){
        shoot.x+=shoot.vx;shoot.y+=shoot.vy;shoot.life++;
        const al=Math.max(0,1-shoot.life/60),tx=shoot.x-shoot.vx*9,ty=shoot.y-shoot.vy*9;
        const gr=ctx.createLinearGradient(shoot.x,shoot.y,tx,ty);
        gr.addColorStop(0,`rgba(255,255,255,${al})`);gr.addColorStop(1,"rgba(255,255,255,0)");
        ctx.strokeStyle=gr;ctx.lineWidth=1.6;ctx.beginPath();ctx.moveTo(shoot.x,shoot.y);ctx.lineTo(tx,ty);ctx.stroke();
        if(shoot.life>60||shoot.x>W||shoot.y>H)shoot=null;
      }
    }
  }
  function loop(t){
    raf=requestAnimationFrame(loop);
    if(t-last<1000/60-1)return; last=t;   // cap at 60fps
    draw(t);
  }
  addEventListener("resize",resize);
  resize();
  if(reduce)return; // static starfield
  addEventListener("scroll",()=>{sy=scrollY;},{passive:true});
  addEventListener("mousemove",e=>{mx=(e.clientX/W-.5)*2;my=(e.clientY/H-.5)*2;},{passive:true});
  document.addEventListener("visibilitychange",()=>{if(document.hidden)cancelAnimationFrame(raf);else raf=requestAnimationFrame(loop);});
  raf=requestAnimationFrame(loop);
})();
